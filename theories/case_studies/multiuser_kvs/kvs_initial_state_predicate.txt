kvs_initial_state_ok (flag : Addr) (σ : ExecConf) :=
  ∃ (sw : img_switcher) (asrt : img_assert) (t0 : img_thread) (main kvs B : img_cmpt) (ot_kvs : OType)
      (key_main key_B : Z),
  -- also, in section B (untrusted, B): ∃ (eh_B ehs_B : Z) (off_B_f min_B_f : nat)
  flag_assert asrt = flag  -- the flag of the assert library
── Parameters ──────────────────────────────────────────────────────────────
  key_main ≠ key_B
── main (trusted) ──────────────────────────────────────────────────────────
  cmpt_imports main =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  (E-Inherit RX, Global, b_assert asrt, e_assert asrt, b_assert asrt)            -- assert
      [2]  { ot_switcher sw,                                                              -- B, export f
             (RO, Global, cmpt_exp_tbl_pcc B, cmpt_exp_tbl_entries_end B, cmpt_exp_tbl_pcc B ^+ ExportTable_offset_entry 0) }
      [3]  { ot_switcher sw,                                                              -- kvs, export addOrUpdate
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 0) }
      [4]  { ot_switcher sw,                                                              -- kvs, export read
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 1) }
      [5]  { ot_switcher sw,                                                              -- kvs, export erase
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 2) }
      [6]  { ot_kvs,                                                                      -- own static user_key
             (RO, Global, cmpt_b_static_sealed main, cmpt_b_static_sealed main ^+ 1, cmpt_b_static_sealed main) }
  cmpt_static_sealed main =
      [0]  WInt key_main
  cmpt_code main                      = kvs_main_code
  cmpt_data main                      = kvs_main_data
  cmpt_exp_tbl_handler main           = WInt ExportTable_no_handler                       -- no error handler
  cmpt_exp_tbl_handler_stackless main = WInt ExportTable_no_handler
  cmpt_exp_tbl_entries main =
      [0]  entry InterruptEnabled 0 6 7                                                   -- export entry
── kvs (trusted) ───────────────────────────────────────────────────────────
  cmpt_imports kvs =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  [U, Global, ot_kvs, (ot_kvs ^+ 1)%ot, ot_kvs]
  cmpt_static_sealed kvs             = []
  cmpt_code kvs                      = kvs_service_instrs
  cmpt_data kvs                      = kvs_data
  cmpt_exp_tbl_handler kvs           = WInt ExportTable_no_handler                        -- no error handler
  cmpt_exp_tbl_handler_stackless kvs = WInt ExportTable_no_handler
  cmpt_exp_tbl_entries kvs =
      [0]  entry InterruptEnabled kvs_addOrUpdate_nargs kvs_addOrUpdate_min_stack kvs_addOrUpdate_pcc_off
                                                                                          -- export addOrUpdate
      [1]  entry InterruptEnabled kvs_read_nargs kvs_read_min_stack kvs_read_pcc_off      -- export read
      [2]  entry InterruptEnabled kvs_erase_nargs kvs_erase_min_stack kvs_erase_pcc_off   -- export erase
── B (untrusted, B) ────────────────────────────────────────────────────────
  cmpt_imports B =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  { ot_switcher sw,                                                              -- kvs, export addOrUpdate
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 0) }
      [2]  { ot_switcher sw,                                                              -- kvs, export read
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 1) }
      [3]  { ot_switcher sw,                                                              -- kvs, export erase
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 2) }
      [4]  { ot_kvs,                                                                      -- own static user_key
             (RO, Global, cmpt_b_static_sealed B, cmpt_b_static_sealed B ^+ 1, cmpt_b_static_sealed B) }
  cmpt_static_sealed B =
      [0]  WInt key_B
  Forall is_int (cmpt_code B)                                                             -- arbitrary, any length
  Forall (initial_data_word (cmpt_b_cgp B) (cmpt_e_cgp B)) (cmpt_data B)                  -- arbitrary, any length
  ∃ eh_B ehs_B : Z,                                                                       -- arbitrary error handler offsets
    cmpt_exp_tbl_handler B = WInt eh_B ∧
    cmpt_exp_tbl_handler_stackless B = WInt ehs_B
  ∃ off_B_f min_B_f : nat,                                                                -- arbitrary offsets, inside the PCC range, and minimum stack sizes
    (off_B_f < cmpt_e_pcc B - cmpt_b_pcc B)%Z ∧
    cmpt_exp_tbl_entries B = [entry InterruptEnabled 0 min_B_f off_B_f]
── Memory ──────────────────────────────────────────────────────────────────
  mem σ =
        initial_heap_memory                                                 -- heap, zeroed
      ∪ mem_region (b_switcher sw)                                          -- switcher
          (switcher_data_words (ot_switcher sw) ++
           switcher_instrs)
      ∪ mem_region (b_assert asrt)                                          -- assert, code and flag capability
          (assert_subroutine_instrs ++

      ∪ mem_region (flag_assert asrt) [WInt 0]                              -- assert, flag
      ∪ mem_region (cmpt_b_pcc main) (cmpt_imports main ++ cmpt_code main)  -- main, PCC range
      ∪ mem_region (cmpt_b_cgp main) (cmpt_data main)                       -- main, CGP range
      ∪ mem_region (cmpt_b_static_sealed main) (cmpt_static_sealed main)    -- main, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc main)                                  -- main, export table
          ((RX, Global, cmpt_b_pcc main, cmpt_e_pcc main, cmpt_b_pcc main) ::
           (RW, Global, cmpt_b_cgp main, cmpt_e_cgp main, cmpt_b_cgp main) ::
           cmpt_exp_tbl_handler main ::

           cmpt_exp_tbl_entries main)
      ∪ mem_region (cmpt_b_pcc kvs) (cmpt_imports kvs ++ cmpt_code kvs)     -- kvs, PCC range
      ∪ mem_region (cmpt_b_cgp kvs) (cmpt_data kvs)                         -- kvs, CGP range
      ∪ mem_region (cmpt_b_static_sealed kvs) (cmpt_static_sealed kvs)      -- kvs, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc kvs)                                   -- kvs, export table

           (RW, Global, cmpt_b_cgp kvs, cmpt_e_cgp kvs, cmpt_b_cgp kvs) ::
           cmpt_exp_tbl_handler kvs ::
           cmpt_exp_tbl_handler_stackless kvs ::
           cmpt_exp_tbl_entries kvs)
      ∪ mem_region (cmpt_b_pcc B) (cmpt_imports B ++ cmpt_code B)           -- B, PCC range
      ∪ mem_region (cmpt_b_cgp B) (cmpt_data B)                             -- B, CGP range
      ∪ mem_region (cmpt_b_static_sealed B) (cmpt_static_sealed B)          -- B, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc B)                                     -- B, export table
          ((RX, Global, cmpt_b_pcc B, cmpt_e_pcc B, cmpt_b_pcc B) ::
           (RW, Global, cmpt_b_cgp B, cmpt_e_cgp B, cmpt_b_cgp B) ::
           cmpt_exp_tbl_handler B ::
           cmpt_exp_tbl_handler_stackless B ::
           cmpt_exp_tbl_entries B)
      ∪ mem_region (b_trusted_stack t0)                                     -- thread 0, trusted stack, entry frame in main
          (initial_trusted_stack_content (RWL, Local, b_stack t0, e_stack t0, (e_stack t0 ^- StackEntry_reserved_space)%f) (RW, Global, cmpt_exp_tbl_pcc main, cmpt_exp_tbl_entries_end main, cmpt_exp_tbl_pcc main) (WO, Global, b_hazard t0, e_hazard t0, b_hazard t0) 24)
      ∪ mem_region (b_stack t0) (replicate 100 (WInt 0))                    -- thread 0, stack
      ∪ mem_region (b_hazard t0) [WInt 0; WInt 0]                           -- thread 0, hazard slots
  shadowtbl σ = initial_heap_shadow                                         -- every heap address live
