kvs_initial_state_ok (flag : Addr) (σ : ExecConf) :=
  ∃ (sw : img_switcher) (asrt : img_assert) (t0 : img_thread) (idle : img_idle_thread)
      (boot : img_boot_stack) (main kvs B sched : img_cmpt) (ot_kvs : OType) (key_main key_B : Z),
  -- also, in section B (untrusted, B): ∃ (eh_B ehs_B : Z) (off_B_f min_B_f : nat)
  flag_assert asrt = flag  -- the flag of the assert library
── Parameters ──────────────────────────────────────────────────────────────
  key_main ≠ key_B
── main (trusted) ──────────────────────────────────────────────────────────
  cmpt_imports main =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  (E-Inherit RX, Global, b_assert asrt, e_assert asrt, b_assert asrt)            -- assert
      [2]  { ot_switcher sw,                                                              -- B, export f
             (RO, Global, cmpt_exp_tbl_pcc B, cmpt_exp_tbl_entries_end B, cmpt_exp_tbl_pcc B ^+ ExportTable_offset_entry 0) }
      [3]  { ot_switcher sw,
          -- kvs, export addOrUpdate
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 0) }
      [4]  { ot_switcher sw,                                                              -- kvs, export read
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 1) }
      [5]  { ot_switcher sw,                                                              -- kvs, export erase
             (RO, Global, cmpt_exp_tbl_pcc kvs, cmpt_exp_tbl_entries_end kvs, cmpt_exp_tbl_pcc kvs ^+ ExportTable_offset_entry 2) }
      [6]  { ot_kvs,                                                                      -- own static user_key
             (RO, Global, cmpt_b_static_sealed main, cmpt_b_static_sealed main ^+ 1, cmpt_b_static_sealed main) }
  cmpt_static_sealed main =
      [0]  WInt key_main
  cmpt_code main                      = kvs_main_code
  cmpt_data main                      = kvs_main_data
  cmpt_exp_tbl_handler main           = WInt ExportTable_no_handler                       -- no error handler
  cmpt_exp_tbl_handler_stackless main = WInt ExportTable_no_handler
  cmpt_exp_tbl_entries main =
      [0]  entry InterruptEnabled 0 6 7                                                   -- export entry

── B (untrusted, B) ────────────────────────────────────────────────────────
  cmpt_imports B =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  { ot_switcher sw,
          -- kvs, export addOrUpdate

      [2]  { ot_switcher sw,                                                              -- kvs, export read

      [4]  { ot_kvs,                                                                      -- own static user_key

      [0]  WInt key_B
  Forall is_int (cmpt_code B)
      -- arbitrary, any length
  Forall (initial_data_word (cmpt_b_cgp B) (cmpt_e_cgp B)) (cmpt_data B)
      -- arbitrary, any length
  ∃ eh_B ehs_B : Z,
      -- arbitrary error handler offsets
    cmpt_exp_tbl_handler B = WInt eh_B ∧
    cmpt_exp_tbl_handler_stackless B = WInt ehs_B
  ∃ off_B_f min_B_f : nat,
      -- arbitrary offsets, inside the PCC range, and minimum stack sizes
    (off_B_f < cmpt_e_pcc B - cmpt_b_pcc B)%Z ∧
    cmpt_exp_tbl_entries B = [entry InterruptEnabled 0 min_B_f off_B_f]

  Forall

── Registers, at the switcher's call of the scheduler ──────────────────────

                             -- the switcher's call of the scheduler

                             -- the scheduler's exception entry, interrupts disabled

                             -- idle thread, handle

                             -- exception entry
