lse_initial_state_ok (flag : Addr) (σ : ExecConf) :=
  ∃ (sw : img_switcher) (assert : img_lib) (assert_flag : img_shared_object) (t0 : img_thread)
      (idle : img_idle_thread) (boot : img_boot_stack) (main C sched : img_cmpt),
  -- also, in section C (untrusted, C): ∃ (eh_C ehs_C : Z) (off_C_adv min_C_adv : nat)
  sh_b assert_flag = flag  -- the assert flag
── main (trusted) ──────────────────────────────────────────────────────────
  cmpt_imports main =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  (E-Inherit RX, Global, lib_b assert, lib_e assert, lib_b assert ^+ 2)
          -- assert, libcall assert
      [2]  { ot_switcher sw,                                                              -- C, export adv
             (RO, Global, cmpt_exp_tbl_pcc C, cmpt_exp_tbl_entries_end C, cmpt_exp_tbl_pcc C ^+ ExportTable_offset_entry 0) }
  cmpt_static_sealed main             = []
  cmpt_code main                      = lse_main_code
  cmpt_data main                      = lse_main_data
  cmpt_exp_tbl_handler main           = WInt ExportTable_no_handler                       -- no error handler
  cmpt_exp_tbl_handler_stackless main = WInt ExportTable_no_handler
  cmpt_exp_tbl_entries main =
      [0]  entry InterruptEnabled 0 1 12                                                  -- export f
      [1]  entry InterruptEnabled 0 4 3                                                   -- export entry
── C (untrusted, C) ────────────────────────────────────────────────────────
  cmpt_imports C =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  { ot_switcher sw,                                                              -- main, export f
             (RO, Global, cmpt_exp_tbl_pcc main, cmpt_exp_tbl_entries_end main, cmpt_exp_tbl_pcc main ^+ ExportTable_offset_entry 0) }
  cmpt_static_sealed C = []
  Forall is_int (cmpt_code C)
      -- arbitrary, any length
  Forall (initial_data_word (cmpt_b_cgp C) (cmpt_e_cgp C)) (cmpt_data C)
      -- arbitrary, any length
  ∃ eh_C ehs_C : Z,
      -- arbitrary error handler offsets
    cmpt_exp_tbl_handler C = WInt eh_C ∧
    cmpt_exp_tbl_handler_stackless C = WInt ehs_C
  ∃ off_C_adv min_C_adv : nat,
      -- arbitrary offsets, inside the PCC range, and minimum stack sizes
    (off_C_adv < cmpt_e_pcc C - cmpt_b_pcc C)%Z ∧
    cmpt_exp_tbl_entries C = [entry InterruptEnabled 0 min_C_adv off_C_adv]
── sched (scheduler) ───────────────────────────────────────────────────────
  cmpt_imports sched =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  (RW_DL_DRO, Global, mtime_addr, mtime_addr ^+ 1, mtime_addr)                   -- time device mtime
      [2]  (RW_DL_DRO, Global, mtimecmp_addr, mtimecmp_addr ^+ 1, mtimecmp_addr)
          -- time device mtimecmp
  cmpt_static_sealed sched = []
  Forall is_int (cmpt_code sched)
      -- arbitrary, any length
  Forall
      -- arbitrary, any length: initial globals, thread handles or imported time-device capabilities
    (sched_data_word (cmpt_b_cgp sched) (cmpt_e_cgp sched)
       [{ ot_switcher sw ^+ 1, (RWL, Global, b_trusted_stack t0, e_trusted_stack t0, b_trusted_stack t0) };
                                                                                          -- thread 0, handle
        { ot_switcher sw ^+ 1,                                                            -- idle thread, handle
          (RWL, Global, img_b_idle_trusted_stack idle, img_e_idle_trusted_stack idle, img_b_idle_trusted_stack idle) };
        (RW_DL_DRO, Global, mtime_addr, mtime_addr ^+ 1, mtime_addr);
        (RW_DL_DRO, Global, mtimecmp_addr, mtimecmp_addr ^+ 1, mtimecmp_addr)])
    (cmpt_data sched)
  cmpt_exp_tbl_handler sched           = WInt ExportTable_no_handler                      -- no error handler
  cmpt_exp_tbl_handler_stackless sched = WInt ExportTable_no_handler
  cmpt_exp_tbl_entries sched           = []
      -- no export: its exception entry is in the switcher
  Forall
      -- the boot stack: integers or thread handles
    (boot_stack_word
       [{ ot_switcher sw ^+ 1, (RWL, Global, b_trusted_stack t0, e_trusted_stack t0, b_trusted_stack t0) };
                                                                                          -- thread 0, handle
        { ot_switcher sw ^+ 1,                                                            -- idle thread, handle
          (RWL, Global, img_b_idle_trusted_stack idle, img_e_idle_trusted_stack idle, img_b_idle_trusted_stack idle) }])
    (img_boot_stack_words boot)
── Memory ──────────────────────────────────────────────────────────────────
  mem σ =
        initial_heap_memory                                                    -- heap, zeroed
      ∪ mem_region (b_switcher sw)                                             -- switcher
          (switcher_data_words
             (ot_switcher sw)                                                    -- ot
             (E-FwdDisable RX, Global, cmpt_b_pcc sched, cmpt_e_pcc sched, cmpt_b_pcc sched ^+ 3)
                 -- the scheduler's exception entry, interrupts disabled
             (RW, Global, cmpt_b_cgp sched, cmpt_e_cgp sched, cmpt_b_cgp sched)  -- the scheduler's globals
             (RWL, Local, img_b_boot_stack boot, img_e_boot_stack boot, img_e_boot_stack boot) ++
                                                                                 -- the boot stack, at its top
           switcher_instrs)
      ∪ mem_region (sh_b assert_flag) (replicate 1 (WInt 0))
          -- shared object assert_flag, zeroed
      ∪ mem_region (lib_b assert) (lib_words assert)                           -- assert, import table and code
      ∪ mem_region (cmpt_b_pcc main) (cmpt_imports main ++ cmpt_code main)     -- main, PCC range
      ∪ mem_region (cmpt_b_cgp main) (cmpt_data main)                          -- main, CGP range
      ∪ mem_region (cmpt_b_static_sealed main) (cmpt_static_sealed main)       -- main, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc main)                                     -- main, export table
          ((RX, Global, cmpt_b_pcc main, cmpt_e_pcc main, cmpt_b_pcc main) ::
           (RW, Global, cmpt_b_cgp main, cmpt_e_cgp main, cmpt_b_cgp main) ::
           cmpt_exp_tbl_handler main ::
           cmpt_exp_tbl_handler_stackless main ::
           cmpt_exp_tbl_entries main)
      ∪ mem_region (cmpt_b_pcc C) (cmpt_imports C ++ cmpt_code C)              -- C, PCC range
      ∪ mem_region (cmpt_b_cgp C) (cmpt_data C)                                -- C, CGP range
      ∪ mem_region (cmpt_b_static_sealed C) (cmpt_static_sealed C)             -- C, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc C)                                        -- C, export table
          ((RX, Global, cmpt_b_pcc C, cmpt_e_pcc C, cmpt_b_pcc C) ::
           (RW, Global, cmpt_b_cgp C, cmpt_e_cgp C, cmpt_b_cgp C) ::
           cmpt_exp_tbl_handler C ::
           cmpt_exp_tbl_handler_stackless C ::
           cmpt_exp_tbl_entries C)
      ∪ mem_region (cmpt_b_pcc sched) (cmpt_imports sched ++ cmpt_code sched)  -- sched, PCC range
      ∪ mem_region (cmpt_b_cgp sched) (cmpt_data sched)                        -- sched, CGP range
      ∪ mem_region (cmpt_b_static_sealed sched) (cmpt_static_sealed sched)     -- sched, static sealed
      ∪ mem_region (cmpt_exp_tbl_pcc sched)                                    -- sched, export table
          ((RX, Global, cmpt_b_pcc sched, cmpt_e_pcc sched, cmpt_b_pcc sched) ::
           (RW, Global, cmpt_b_cgp sched, cmpt_e_cgp sched, cmpt_b_cgp sched) ::
           cmpt_exp_tbl_handler sched ::
           cmpt_exp_tbl_handler_stackless sched ::
           cmpt_exp_tbl_entries sched)
      ∪ mem_region (b_trusted_stack t0)
          -- thread 0, trusted stack, saved at its entry in main
          (initial_trusted_stack_content
             (save_area_content                                                                  -- save
                (RX, Global, cmpt_b_pcc main, cmpt_e_pcc main, cmpt_b_pcc main ^+ 3)             -- mepcc
                (E-RetDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_return sw)  -- cra
                (RWL, Local, b_stack t0, e_stack t0, e_stack t0 ^- StackEntry_reserved_space)    -- csp
                (RW, Global, cmpt_b_cgp main, cmpt_e_cgp main, cmpt_b_cgp main)                  -- cgp
                (WO, Global, b_hazard t0, e_hazard t0, b_hazard t0)                              -- hazard
                (WInt (mstatus_enc InterruptDisabled InterruptEnabled)))                         -- mstatus
             (RWL, Local, b_stack t0, e_stack t0, e_stack t0 ^- StackEntry_reserved_space)       -- csp
             (RW, Global, cmpt_exp_tbl_pcc main, cmpt_exp_tbl_entries_end main, cmpt_exp_tbl_pcc main)
                                                                                                 -- export_table
             1                                                                                   -- thread_id
             24)                                                                                 -- n
      ∪ mem_region (b_stack t0) (replicate 100 (WInt 0))                       -- thread 0, stack
      ∪ mem_region (b_hazard t0) [WInt 0; WInt 0]                              -- thread 0, hazard slots
      ∪ mem_region (img_b_idle_trusted_stack idle)
          -- idle thread, trusted stack, saved at the loop of an exited thread
          (initial_trusted_stack_content
             (save_area_content                                                -- save
                (RX,                                                           -- mepcc
                 Local,
                 a_switcher_call sw ^+ label ".Lexited_thread",
                 (a_switcher_call sw ^+ label ".Lexited_thread") ^+ 2,
                 a_switcher_call sw ^+ label ".Lexited_thread")
                (WInt 0)                                                       -- cra
                (WInt 0)                                                       -- csp
                (WInt 0)                                                       -- cgp
                (WO, Global, img_b_idle_hazard idle, img_e_idle_hazard idle, img_b_idle_hazard idle)
                                                                               -- hazard
                (WInt (mstatus_enc InterruptDisabled InterruptEnabled)))       -- mstatus
             (WInt 0)                                                          -- csp
             (WInt 0)                                                          -- export_table
             0                                                                 -- thread_id
             0)                                                                -- n
      ∪ mem_region (img_b_idle_hazard idle) [WInt 0; WInt 0]                   -- idle thread, hazard slots
      ∪ mem_region (img_b_boot_stack boot) (img_boot_stack_words boot)
          -- boot stack, the scheduler's stack
  shadowtbl σ = initial_heap_shadow                                            -- every heap address live
── Registers, at the switcher's call of the scheduler ──────────────────────
  pcc σ = (XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw ^+ label ".Lexception_scheduler_cjalr")
                             -- the switcher's call of the scheduler
  reg σ !! cra = Some (E-FwdDisable RX, Global, cmpt_b_pcc sched, cmpt_e_pcc sched, cmpt_b_pcc sched ^+ 3)
                             -- the scheduler's exception entry, interrupts disabled
  reg σ !! cgp = Some (RW, Global, cmpt_b_cgp sched, cmpt_e_cgp sched, cmpt_b_cgp sched)
                             -- the scheduler's globals
  reg σ !! csp = Some (RWL, Local, img_b_boot_stack boot, img_e_boot_stack boot, img_e_boot_stack boot)
                             -- the boot stack, at its top
  reg σ !! ca0 = Some { ot_switcher sw ^+ 1,
                        (RWL, Global, img_b_idle_trusted_stack idle, img_e_idle_trusted_stack idle, img_b_idle_trusted_stack idle) }
                             -- idle thread, handle
  ∃ z1 z2 z3 : Z,            -- arbitrary integers
    reg σ !! ca1 = Some (WInt z1)
    reg σ !! ca2 = Some (WInt z2)
    reg σ !! ca3 = Some (WInt z3)
  ∀ r : RegName,             -- every other register cleared
    r ∉ {[cra; cgp; csp; ca0; ca1; ca2; ca3]} → reg σ !! r = Some (WInt 0)
  sreg σ !! mtdc = Some (WInt 0)
  sreg σ !! mtcc = Some (XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw ^+ label "exception_entry_asm")
                             -- exception entry
  ∃ mpie : InterruptStatus,  -- interrupts disabled, MPIE arbitrary
    sreg σ !! mstatus = Some (WInt (mstatus_enc InterruptDisabled mpie))
  ∃ z1 z2 z3 : Z,            -- arbitrary integers, dead until the resume path
    sreg σ !! mepcc  = Some (WInt z1)
    sreg σ !! mcause = Some (WInt z2)
    sreg σ !! mtval  = Some (WInt z3)
── switcher, layout ────────────────────────────────────────────────────────
  b_switcher sw + 5                                 = Some (a_switcher_call sw)
  a_switcher_call sw + length switcher_instrs       = Some (e_switcher sw)
  b_switcher sw + (5 + length switcher_call_instrs) = Some (a_switcher_return sw)
  not_heap_range (b_switcher sw) (e_switcher sw)
  disjoint_from_shadow (b_switcher sw) (e_switcher sw)
  revoker_addr ∉ finz.seq_between (b_switcher sw) (e_switcher sw)
  mtime_addr ∉ finz.seq_between (b_switcher sw) (e_switcher sw)
  mtimecmp_addr ∉ finz.seq_between (b_switcher sw) (e_switcher sw)
  is_heap_address (e_switcher sw) = false  -- closed bounds outside the heap
── shared object assert_flag, layout ───────────────────────────────────────
  sh_b assert_flag + 1%nat = Some (sh_e assert_flag)
  not_heap_range (sh_b assert_flag) (sh_e assert_flag)
  disjoint_from_shadow (sh_b assert_flag) (sh_e assert_flag)
  revoker_addr ∉ finz.seq_between (sh_b assert_flag) (sh_e assert_flag)
  mtime_addr ∉ finz.seq_between (sh_b assert_flag) (sh_e assert_flag)
  mtimecmp_addr ∉ finz.seq_between (sh_b assert_flag) (sh_e assert_flag)
── assert (library) ────────────────────────────────────────────────────────
  lib_words assert =
      [0]  (E-FwdDisable XSRW_, Local, b_switcher sw, e_switcher sw, a_switcher_call sw)  -- switcher call
      [1]  (RW, Global, sh_b assert_flag, sh_e assert_flag, sh_b assert_flag)
          -- shared object assert_flag
      ++ assert_subroutine_instrs
── assert, layout ──────────────────────────────────────────────────────────
  lib_b assert + (2 + length assert_subroutine_instrs)%nat = Some (lib_e assert)
  not_heap_range (lib_b assert) (lib_e assert)
  disjoint_from_shadow (lib_b assert) (lib_e assert)
  revoker_addr ∉ finz.seq_between (lib_b assert) (lib_e assert)
  mtime_addr ∉ finz.seq_between (lib_b assert) (lib_e assert)
  mtimecmp_addr ∉ finz.seq_between (lib_b assert) (lib_e assert)
  is_heap_address (lib_e assert) = false     -- closed bounds outside the heap
  2 ≤ 2                                      -- export assert in the code
  (2 < 2 + length assert_subroutine_instrs)%nat
  Forall (word_rooted ∅) (lib_words assert)  -- no heap capability (no heap root)
  Forall word_exec_ok (lib_words assert)     -- executable words outside the heap
── main, layout ────────────────────────────────────────────────────────────
  cmpt_b_pcc main + length (cmpt_imports main)                         = Some (cmpt_a_code main)

  not_heap_range (cmpt_b_static_sealed main) (cmpt_e_static_sealed main)  -- main, static sealed

  mtimecmp_addr ∉ finz.seq_between (cmpt_b_static_sealed main) (cmpt_e_static_sealed main)
  not_heap_range (cmpt_exp_tbl_pcc main) (cmpt_exp_tbl_entries_end main)  -- main, export table
  disjoint_from_shadow (cmpt_exp_tbl_pcc main) (cmpt_exp_tbl_entries_end main)
  revoker_addr ∉ finz.seq_between (cmpt_exp_tbl_pcc main) (cmpt_exp_tbl_entries_end main)
  mtime_addr ∉ finz.seq_between (cmpt_exp_tbl_pcc main) (cmpt_exp_tbl_entries_end main)
  mtimecmp_addr ∉ finz.seq_between (cmpt_exp_tbl_pcc main) (cmpt_exp_tbl_entries_end main)
  is_heap_address (cmpt_e_pcc main) = false                               -- closed PCC bounds outside the heap
  (12%nat < cmpt_e_pcc main - cmpt_b_pcc main)%Z                          -- export f inside the PCC range
  (3%nat < cmpt_e_pcc main - cmpt_b_pcc main)%Z                           -- export entry inside the PCC range
  Forall (word_rooted ∅)                                                  -- no heap capability (no heap root)
    (cmpt_imports main ++
     cmpt_code main ++
     cmpt_data main ++
     cmpt_static_sealed main ++

     cmpt_exp_tbl_entries main)
